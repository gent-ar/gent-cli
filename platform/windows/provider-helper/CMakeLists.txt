cmake_minimum_required(VERSION 3.20)
project(gent_windows_sandbox LANGUAGES CXX)
add_executable(gent-windows-sandbox main.cpp command_broker.cpp readonly_authority.cpp filesystem_aliases.cpp gateway_proxy.cpp gateway_proxy_server.cpp resource_inspection.cpp)
set_property(TARGET gent-windows-sandbox PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_features(gent-windows-sandbox PRIVATE cxx_std_17)
target_compile_definitions(gent-windows-sandbox PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(gent-windows-sandbox PRIVATE /W4 /WX /guard:cf /sdl)
target_link_options(gent-windows-sandbox PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /DEPENDENTLOADFLAG:0x800)
target_link_libraries(gent-windows-sandbox PRIVATE bcrypt userenv advapi32 ws2_32 shell32 ole32 OneCoreUAP wintrust crypt32 sfc iphlpapi)
add_executable(gent-windows-sandbox-probe probe.cpp probe_network.cpp probe_resources.cpp)
set_property(TARGET gent-windows-sandbox-probe PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_features(gent-windows-sandbox-probe PRIVATE cxx_std_17)
target_compile_definitions(gent-windows-sandbox-probe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(gent-windows-sandbox-probe PRIVATE /W4 /WX /guard:cf /sdl)
target_link_options(gent-windows-sandbox-probe PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /DEPENDENTLOADFLAG:0x800)
target_link_libraries(gent-windows-sandbox-probe PRIVATE ws2_32 advapi32)

# The host adapter carries an explicit reviewed package authority. Missing pins
# remain fail-closed; this target does not enable the daemon policy allowlist.
add_subdirectory(external-package)
target_link_libraries(gent-windows-sandbox PRIVATE gent-external-package-host)

    target_compile_definitions(gent-filesystem-alias-test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    target_compile_options(gent-filesystem-alias-test PRIVATE /W4 /WX /guard:cf /sdl)
    target_link_options(gent-filesystem-alias-test PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT)
    file(TO_NATIVE_PATH "${CMAKE_CURRENT_BINARY_DIR}/filesystem-alias-owned" GENT_FILESYSTEM_ALIAS_TEST_ROOT)

endif()
