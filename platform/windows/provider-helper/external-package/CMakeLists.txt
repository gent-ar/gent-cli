cmake_minimum_required(VERSION 3.20)
project(gent_external_package_proxy LANGUAGES CXX)
add_executable(gent-network-proxy WIN32 package_main.cpp package_server.cpp packaged_http_forwarder.cpp)
set_property(TARGET gent-network-proxy PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_features(gent-network-proxy PRIVATE cxx_std_17)
target_compile_definitions(gent-network-proxy PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(gent-network-proxy PRIVATE /W4 /WX /guard:cf /sdl)
target_link_options(gent-network-proxy PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /DEPENDENTLOADFLAG:0x800)
target_link_libraries(gent-network-proxy PRIVATE ws2_32 advapi32 shell32)

# Host adapter links into the reviewed helper; no hardcoded QA identity/pin.
add_library(gent-external-package-host STATIC external_package_proxy.cpp package_authority.cpp
    package_owner.cpp package_control.cpp package_peer.cpp host_pipe_bridge.cpp)
set_property(TARGET gent-external-package-host PROPERTY MSVC_RUNTIME_LIBRARY MultiThreaded)
target_compile_features(gent-external-package-host PRIVATE cxx_std_17)
target_include_directories(gent-external-package-host PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_definitions(gent-external-package-host PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
target_compile_options(gent-external-package-host PRIVATE /W4 /WX /guard:cf /sdl)
target_link_libraries(gent-external-package-host PRIVATE bcrypt userenv advapi32 ws2_32 shell32 ole32 OneCoreUAP uuid)

set(GENT_EXTERNAL_PACKAGE_FAMILY "" CACHE STRING "Reviewed installed proxy package family")
set(GENT_EXTERNAL_PACKAGE_FULL_NAME "" CACHE STRING "Reviewed exact installed proxy package full name")
set(GENT_EXTERNAL_PACKAGE_IMAGE_SHA256 "" CACHE STRING "Reviewed proxy payload SHA256")
set(GENT_EXTERNAL_PACKAGE_MANIFEST_SHA256 "" CACHE STRING "Reviewed AppxManifest.xml SHA256")
set(GENT_EXTERNAL_PACKAGE_APPLICATION "GentNetworkProxy" CACHE STRING "Reviewed package application id")
set(GENT_EXTERNAL_PACKAGE_IMAGE [=[data\gent-network-proxy\gent-network-proxy.exe]=] CACHE STRING "Reviewed package-relative proxy image")
option(GENT_EXTERNAL_PACKAGE_ALLOW_DEVELOPMENT_FOR_TEST "DIAGNOSTIC ONLY: permit reviewed loose QA package" OFF)
set(authority_present 0)
foreach(field FAMILY FULL_NAME IMAGE_SHA256 MANIFEST_SHA256)
  if(NOT "${GENT_EXTERNAL_PACKAGE_${field}}" STREQUAL "")
    math(EXPR authority_present "${authority_present}+1")
  endif()
endforeach()
if(NOT authority_present EQUAL 0 AND NOT authority_present EQUAL 4)
  message(FATAL_ERROR "External package authority must be complete; partial pins are rejected")
endif()
if(authority_present EQUAL 4)
  foreach(field FAMILY FULL_NAME APPLICATION)
    if(NOT "${GENT_EXTERNAL_PACKAGE_${field}}" MATCHES "^[A-Za-z0-9_.-]+$")
      message(FATAL_ERROR "Invalid external package ${field}")
    endif()
    target_compile_definitions(gent-external-package-host PRIVATE "GENT_EXTERNAL_PACKAGE_${field}=L\"${GENT_EXTERNAL_PACKAGE_${field}}\"")
  endforeach()
  foreach(field IMAGE_SHA256 MANIFEST_SHA256)
    string(LENGTH "${GENT_EXTERNAL_PACKAGE_${field}}" length)
    if(NOT length EQUAL 64 OR "${GENT_EXTERNAL_PACKAGE_${field}}" MATCHES "[^a-fA-F0-9]")
      message(FATAL_ERROR "Invalid external package ${field}")
    endif()
    string(TOLOWER "${GENT_EXTERNAL_PACKAGE_${field}}" digest)
    target_compile_definitions(gent-external-package-host PRIVATE "GENT_EXTERNAL_PACKAGE_${field}=\"${digest}\"")
  endforeach()
  string(REPLACE "\\" "/" normalized_image "${GENT_EXTERNAL_PACKAGE_IMAGE}")
  if(NOT normalized_image MATCHES "^[A-Za-z0-9_.-]+(/[A-Za-z0-9_.-]+)*$" OR normalized_image MATCHES "(^|/)\\.\\.?(/|$)")
    message(FATAL_ERROR "External package image must be a reviewed relative path")
  endif()
  string(REPLACE "/" "\\\\" escaped_image "${normalized_image}")
  target_compile_definitions(gent-external-package-host PRIVATE "GENT_EXTERNAL_PACKAGE_IMAGE=L\"${escaped_image}\"")
endif()
if(GENT_EXTERNAL_PACKAGE_ALLOW_DEVELOPMENT_FOR_TEST)
  message(WARNING "DIAGNOSTIC external proxy authority: not production mode acceptance")
  target_compile_definitions(gent-external-package-host PRIVATE GENT_EXTERNAL_PACKAGE_ALLOW_DEVELOPMENT_FOR_TEST=1)
endif()
